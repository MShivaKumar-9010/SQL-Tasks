1.Create a stored procedure to display all records from a table called Employees.
   delimiter //
   create procedure emp()
   begin
   select * from employees;
   end//
   call emp() //

2.Write a stored procedure that accepts an EmployeeID and returns the employee details.
   delimiter //
   create procedure emp1(in emp_id int )
   begin
   select * from employees where employeeID=emp_id;
   end //
   call emp1(3)//

3.Create a stored procedure that inserts a new employee into the Employees table with parameters:
* Name
* Department
* Salary
   delimiter //
   create procedure emp2()
   begin
   insert into employ values("Hasini","HR",50000);
   end//
   call emp2()//

4.Write a stored procedure to update the salary of an employee using EmployeeID.
   delimiter //
   create procedure emp3()
   begin
   update employ set salary=80000 where name='priya';
   end//
   call emp3()//

5.Create a stored procedure to delete an employee record using EmployeeID.
   delimiter //
   create procedure emp4()
   begin
   delete from employ where name='Suresh';
   end//
   call emp4()//

6.Write a stored procedure that takes Salary as input and displays employees whose salary is greater than that value.
   delimiter //
   create procedure emp5(in emp_salary int)
   begin
   select * from employ where salary>emp_salary; 
   end//
   call emp5(50000)//

7.Write a stored procedure to calculate the average salary of employees.
   delimiter //
   create procedure emp6()
   begin
   select avg(salary) from employ;
   end//
   call emp6()//

8.Create a stored procedure that returns the total number of employees in a department.
   delimiter //
   create procedure emp7()
   begin
   select count(department) from employ group by department;
   end//
   call emp7()//

9.Write a stored procedure to find the maximum salary in the Employees table.
   delimiter //
   create procedure emp8()
   begin
   select max(salary) from employ;
   end//
   call emp8()//

10.Create a stored procedure that gives a bonus (10%) to employees whose salary is below 50,000.
   delimiter //
   create procedure bonus() 
   begin 
   update employ set salary=salary+(salary*10/100) where salary<50000; 
   end//
   call bonus()//
