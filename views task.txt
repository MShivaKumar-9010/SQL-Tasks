TASK ON VIEWS:
1. Create a view named emp_view to display all employee details.
create view emp_view as
select * from employees;
select * from emp_view;
2. Create a view named emp_basic to display only employee ID, name and department.
create view emp_basic as
select emp_id,emp_name,dept_name from employees;
select * from emp_basic;
3. Create a view named hr_employees to display employees who belong to the HR department.
 create view hr_employees as
 select * from employees where dept_name="HR";
 select * from hr_employees;
4. Create a view named high_salary to display employees whose salary is greater than 50,000.
create view high_salary as
select * from employees where salary>50000;
select * from high_salary;
5. Display all records from the emp_view view.
create view emp_view as
select * from employees;
select * from emp_view;
6. Display only employee name and salary from the emp_view view.
create view emp_sal as
select emp_name,salary from emp_view;
select * from emp_sal;
7. Show employees who belong to the Hyderabad city using a view.
create view city as
select * from employees where city="hyderabad";
select * from city;
8. Create a view named dept_salary that displays department name and average salary of each department.
create view dept_salary as
select dept_name,avg(salary) from employees group by dept_name;
select * from dept_salary;
9. Create a view named experienced_employees to display employees having more than 3 years of experience.
create view experienced_employees as
select * from employees where exp_years>3;
select * from experienced_employees;
10. Create a view named salary_details to display employee name, department, salary and salary category.
Salary category:
- Salary greater than 60,000 → High
- Salary between 30,000 and 60,000 → Medium
- Salary below 30,000 → Low
create view salary_details as
select emp_name,dept_name,salary,
case
when salary>60000 then"high"
when salary between 30000 and 60000 then "medium"
when salary<30000 then "low"
end as salary_category
from employees;
select * from salary_details;
11. Create a view that displays employees whose salary is greater than the average salary of all employees.
create view max_salary as 
select * from employees where salary>(select avg(salary) from employees);
select * from max_salary;
12. Create a view that displays the highest-paid employee in each department.
create view high_paid as
select emp_name,dept_name,max(salary) from employees group by dept_name;
select * from high_paid;
13. Create a view that displays department-wise employee count.
create view count as
select count(emp_name),dept_name from employees group by dept_name;
select * from count;
14. Create a view to display employees whose names start with the letter 'A'.
create view letter as 
select * from employees where emp_name like "A%";
select * from letter;
